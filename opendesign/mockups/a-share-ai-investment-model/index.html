模型伪代码

Pseudocode
universe = filter_a_share(
  exclude_st=True,
  min_amount_20d=300_000_000,
  listed_days_gt=250
)

features = build_features(
  price_window=[20, 60, 120],
  financial_quarters=8,
  event_docs=latest_180d,
  industry_neutral=True
)

alpha_score =
  0.30 * growth_acceleration +
  0.15 * quality_gate +
  0.25 * moneyflow_confirmation +
  0.20 * price_breakout +
  0.10 * ai_event_score

risk_block = liquidity_risk
  or crowding_score > 0.85
  or event_risk in ["监管问询", "大额解禁"]

signal = rank(alpha_score)
  .where(alpha_score > 75)
  .where(not risk_block)
  .top_n(12)
